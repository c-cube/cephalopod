  2. | No_errors : unit errors

  1. pp : 'e pp;
  2. to_value : 'e -> Cephalopod_dasl.Value.t;

type 'a input_or_output =
  1. | IO_encodable : {
    1. encoding : encoding;
    2. encode : 'a encodable;
    } -> 'a input_or_output
  2. | IO_opaque : {
    1. encoding : encoding;
    } -> unit input_or_output
  3. | No_io : unit input_or_output
type ('params, 'o, 'errors) query = {
  1. parameters : 'params params;
  2. output : 'o input_or_output;
  3. errors : 'errors errors;
}
